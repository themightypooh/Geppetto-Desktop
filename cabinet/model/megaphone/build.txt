# Novelty chat button 2: a brass speaking trumpet / megaphone, axis along X.

echo handle: dark wood grip under the middle of the horn
add Blob name=Handle voxel=0.03 blend=0.01 shapes="box -0.3,0,0.45 0.45,0.55,0.85 r=0.08 mat=1"
describe
turnaround out shade=material size=512 views=front,quarter,side,back name=megaphone
save megaphone.effigy
export out name=megaphone
